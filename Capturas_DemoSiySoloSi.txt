Variables de la pregunta
------------------------------
stack_include("contribl://prooflib.mac");
ta:proof_iff(proof("assodd","defn_odd","alg_odd","def_M_odd","conc_odd"), proof("contrapos","assnotodd","even","alg_even","def_M_even","conc_even"));

proof_steps: [
    ["assodd",     "Asumamos que \\(n\\) es impar."],
    ["defn_odd",   "Entonces, existe \\(m\\in\\mathbb{Z}\\) tal que \\(n=2m+1\\)."],
    ["alg_odd",    "\\[ n^2 = (2m+1)^2 = 2(2m^2+2m)+1.\\]"],
    ["def_M_odd",  "Definamos \\(M=2m^2+2m\\in\\mathbb{Z}\\). Entonces \\(n^2=2M+1\\)."],
    ["conc_odd",   "Por lo tanto, \\(n^2\\) es impar."],

    ["contrapos",  "Reformulamos ahora \"\\(n^2\\) es impar \\(\\Rightarrow \\) \\(n\\) es impar \" como contrapositiva."],
    ["assnotodd",  "Asumamos que \\(n\\) no es impar."],
    ["even",       "Entonces \\(n\\) es par, y existe \\(m\\in\\mathbb{Z}\\) tal que \\(n=2m\\)."],
    ["alg_even",   "\\[ n^2 = (2m)^2 = 2(2m^2).\\]"],
    ["def_M_even", "Definamos \\(M=2m^2\\in\\mathbb{Z}\\). Entonces, \\(n^2=2M\\)."],
    ["conc_even",  "Por lo tanto, \\(n^2\\) es par."]
];

/* Generación de demostraciones alternativas. */
tal:proof_alternatives(ta);
/* Creación de un conjunto de pruebas "aplanadas"*/
tas:setify(map(proof_flatten, tal));
proof_steps:random_permutation(proof_steps);
